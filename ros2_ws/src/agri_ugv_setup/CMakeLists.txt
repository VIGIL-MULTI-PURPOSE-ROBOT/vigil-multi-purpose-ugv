cmake_minimum_required(VERSION 3.16)
project(agri_ugv_setup)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(transport_smoke src/transport_smoke.cpp)
target_compile_features(transport_smoke PUBLIC cxx_std_17)
ament_target_dependencies(transport_smoke rclcpp std_msgs)
install(TARGETS transport_smoke DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/check_environment.py DESTINATION lib/${PROJECT_NAME})
ament_package()
