cmake_minimum_required(VERSION 3.16)
project(sar_dynamic_obstacles LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
add_library(sar-waypoint-system SHARED plugins/WaypointSystem.cc)
target_compile_features(sar-waypoint-system PRIVATE cxx_std_17)
target_link_libraries(sar-waypoint-system PRIVATE gz-sim8::gz-sim8)
target_compile_options(sar-waypoint-system PRIVATE -Wall -Wextra)
include(CTest)
if(BUILD_TESTING)
  add_executable(check_motion tests/check_motion.cc)
  target_link_libraries(check_motion PRIVATE gz-sim8::gz-sim8)
  target_compile_features(check_motion PRIVATE cxx_std_17)
  add_executable(check_contacts tests/check_contacts.cc)
  target_link_libraries(check_contacts PRIVATE gz-sim8::gz-sim8)
  target_compile_features(check_contacts PRIVATE cxx_std_17)
endif()
